#include <bits/stdc++.h>
using namespace std;

void frequency(int arr[], int n){
    unordered_map<int, int> map;
    for(int i=0 ; i < n ; i++){
        map[arr[i]]++;
    }
    for (auto x : map){
        cout << x.first << "->" << x.second << endl;
    }
}
int main(){
    int arr[] = {10, 5, 3, 36, 73, 73, 56, 5};
    int n = sizeof(arr) / sizeof(arr[0]);  // it is used to find the size(number of elements) sizeof(arr)-> calculates the byte size of whole array while sizeof(arr[0])-> calculates the bytes occupied by one element
    
    frequency(arr, n);
    return 0;
}
